无人飞机决策系统

一。c++（运算符，输入输出）
干去研究知识还是太无趣了，就从一个无人机起飞前的检测系统来把知识穿起来
就是希望传感器的数据输入后计算机能够做出良好的决策
#include <iostream>#include <iomanip>using namespace std;
int main(){    // 1. 输入无人机状态    double voltage;        // 电池电压 V    double current;        // 当前电流 A    double capacity;       // 剩余容量 Ah    double temperature;    // 电池温度 ℃
    int gpsSatellites;     // GPS卫星数    int motorState;        // 电机状态位    int flightNumber;      // 飞行次数
    cout << "===== Drone Pre-flight Check =====" << endl;
    cout << "Battery voltage (V): ";    cin >> voltage;
    cout << "Current (A): ";    cin >> current;
    cout << "Remaining capacity (Ah): ";    cin >> capacity;
    cout << "Battery temperature (C): ";    cin >> temperature;
    cout << "GPS satellites: ";    cin >> gpsSatellites;
    cout << "Motor state (0~15): ";    cin >> motorState;
    cout << "Flight number: ";    cin >> flightNumber;
    // 2. 算术运算
    double power = voltage * current;      // 定义功率(double运算后还是double，就按精度大的来)    // 这里的运算结果要看操作数数值，类型和运算符规则        double estimatedTime =        current > 0 ? capacity / current * 60 : 0;    // 条件表达式(看电流是否大于零，true就直接求时间，false就不能飞了，少有的三目运算符)       // 3. 关系表达式    bool voltageOK = voltage >= 14.0;    bool temperatureOK =        temperature >= 0 && temperature <= 60;    // 这里的&&实际上同时承担两个职责，逻辑判断和控制右边代码要不要执行    bool gpsOK = gpsSatellites >= 6;       // 4. 位运算    // motorState:无人飞机的四个电机，bit0 = motor1，bit1 = motor2，bit2 = motor3，bit3 = motor4    // 1 = 正常(这里是化为16进制后，硬件系统经常会直接给你一个状态寄存器    // 00001111,一号电机就是看motorstate，二号电机就要进行一个掩码的操作了)    int motor1 = motorState & 1;    int motor2 = (motorState >> 1) & 1;    int motor3 = (motorState >> 2) & 1;    int motor4 = (motorState >> 3) & 1;
    bool motorsOK =        motor1 && motor2 && motor3 && motor4;
    // 5. 逻辑运算 + 短路    bool enduranceOK =        current > 0 && capacity / current > 0.15;
    // 6. 最终起飞判断    bool canFly =    voltageOK && temperatureOK && gpsOK && motorsOK && enduranceOK;
    // 7. 三目运算符    const char* result =        canFly ? "READY TO FLY" : "DO NOT TAKE OFF";
    // 8. 自增(就是一个状态的改变)    int checkCount = 0;
    cout << "\nCheck ID: " << checkCount++ << endl;    cout << "Next ID : " << checkCount << endl;
    // 9. % 取余(这里=和==是不一样的，前者是赋值，后者是等于)    // 每10次飞行维护一次    bool needMaintenance =        flightNumber % 10 == 0;// 要看数据类型！
    // 10. 类型转换(可以主动丢失信息也可提高精确度)    int integerPower = (int)power;
    // 11. 格式化输出    cout << fixed << setprecision(2);
    cout << "\n===== Flight Report =====" << endl;
    cout << "Voltage : "         << voltage << " V" << endl;
    cout << "Power   : "         << power << " W" << endl;
    cout << "Power(int): "         << integerPower << " W" << endl;
    cout << "Estimated flight time: "         << estimatedTime << " min" << endl;
    // 12. 十六进制输出    cout << "Motor state: 0x"         << hex << motorState         << dec << endl;
    // 13. sizeof    cout << "sizeof(voltage): "         << sizeof(voltage)         << " bytes" << endl;
    // 14. 三目运算再次使用    cout << "Maintenance: "         << (needMaintenance ? "YES" : "NO")         << endl;
    cout << "\nFINAL RESULT: "         << result << endl;
    return 0;}
